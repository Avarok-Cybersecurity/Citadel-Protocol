1//! The transport offer exchange that makes both ends of a media session agree
2//! on UDP or the reliable lane.
3use super::error::MediaResultExt;
4use super::transport::{BoxedSource, MediaDatagramSink};
5use bytes::BytesMut;
6use citadel_io::ErrorCode;
7use citadel_media::wire::{self, encode_control, WireMessage};
8use citadel_media::ControlMessage;
9use citadel_proto::constants::{protocol_version_at_least, MEDIA_TRANSPORT_OFFER_SINCE};
10use citadel_proto::prelude::NetworkError;
11use futures::StreamExt;
12
13/// Whether media uses UDP: only if this side holds it and the peer's offer
14/// says it does too. A peer whose protocol version predates transport offers,
15/// or is unknown, sends none and reads none, so it is sent none and never
16/// waited on: the answer is reliable.
17pub(super) async fn agree_on_udp(
18    peer_protocol_version: Option<u32>,
19    local_udp: bool,
20    sink: &mut dyn MediaDatagramSink,
21    src: &mut BoxedSource,
22) -> Result<bool, NetworkError> {
23    if !protocol_version_at_least(peer_protocol_version, MEDIA_TRANSPORT_OFFER_SINCE) {
24        log::warn!(target: "citadel", "media: peer protocol version {peer_protocol_version:?} predates transport offers; using reliable transport");
25        return Ok(false);
26    }
27    let peer_udp = exchange_transport_offer(sink, src, local_udp).await?;
28    Ok(local_udp && peer_udp)
29}
30
31/// Sends this side's transport offer as the first reliable message and reads
32/// the peer's, which is likewise the first message on the ordered lane.
33/// Returns whether the peer holds UDP. Each side's own UDP decision is local
34/// and timer-bound, so only the pair of offers yields an agreed transport.
35async fn exchange_transport_offer(
36    sink: &mut dyn MediaDatagramSink,
37    src: &mut BoxedSource,
38    local_udp: bool,
39) -> Result<bool, NetworkError> {
40    let body = ControlMessage::TransportOffer { udp: local_udp }
41        .encode()
42        .net()?;
43    sink.send_datagram(BytesMut::from(encode_control(&body).as_slice()))?;
44    let reply = src.next().await.ok_or_else(|| {
45        citadel_io::error!(
46            ErrorCode::MediaTransportClosed,
47            "reliable channel closed before the peer's transport offer"
48        )
49    })?;
50    let malformed = |err| citadel_io::error!(ErrorCode::MediaControlDecode, err);
51    match wire::parse(reply.as_ref()).map_err(malformed)? {
52        WireMessage::Control(body) => match ControlMessage::decode(body).map_err(malformed)? {
53            ControlMessage::TransportOffer { udp } => Ok(udp),
54            other => Err(citadel_io::error!(
55                ErrorCode::MediaControlDecode,
56                format!("expected the peer's transport offer, got {other:?}")
57            )),
58        },
59        WireMessage::Fragment { .. } => Err(citadel_io::error!(
60            ErrorCode::MediaControlDecode,
61            "expected the peer's transport offer, got a media fragment"
62        )),
63    }
64}
