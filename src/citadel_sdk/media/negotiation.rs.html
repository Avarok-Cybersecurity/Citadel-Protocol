1//! The transport offer exchange that makes both ends of a media session agree
2//! on UDP or the reliable lane.
3use super::error::MediaResultExt;
4use super::transport::{BoxedSource, MediaDatagramSink};
5use bytes::BytesMut;
6use citadel_io::ErrorCode;
7use citadel_media::wire::{self, encode_control, WireMessage};
8use citadel_media::ControlMessage;
9use citadel_proto::prelude::NetworkError;
10use futures::StreamExt;
11
12/// Whether media uses UDP: only if this side holds it and the peer's offer
13/// says it does too.
14pub(super) async fn agree_on_udp(
15    local_udp: bool,
16    sink: &mut dyn MediaDatagramSink,
17    src: &mut BoxedSource,
18) -> Result<bool, NetworkError> {
19    let peer_udp = exchange_transport_offer(sink, src, local_udp).await?;
20    Ok(local_udp && peer_udp)
21}
22
23/// Sends this side's transport offer as the first reliable message and reads
24/// the peer's, which is likewise the first message on the ordered lane.
25/// Returns whether the peer holds UDP. Each side's own UDP decision is local
26/// and timer-bound, so only the pair of offers yields an agreed transport.
27async fn exchange_transport_offer(
28    sink: &mut dyn MediaDatagramSink,
29    src: &mut BoxedSource,
30    local_udp: bool,
31) -> Result<bool, NetworkError> {
32    let body = ControlMessage::TransportOffer { udp: local_udp }
33        .encode()
34        .net()?;
35    sink.send_datagram(BytesMut::from(encode_control(&body).as_slice()))?;
36    let reply = src.next().await.ok_or_else(|| {
37        citadel_io::error!(
38            ErrorCode::MediaTransportClosed,
39            "reliable channel closed before the peer's transport offer"
40        )
41    })?;
42    let malformed = |err| citadel_io::error!(ErrorCode::MediaControlDecode, err);
43    match wire::parse(reply.as_ref()).map_err(malformed)? {
44        WireMessage::Control(body) => match ControlMessage::decode(body).map_err(malformed)? {
45            ControlMessage::TransportOffer { udp } => Ok(udp),
46            other => Err(citadel_io::error!(
47                ErrorCode::MediaControlDecode,
48                format!("expected the peer's transport offer, got {other:?}")
49            )),
50        },
51        WireMessage::Fragment { .. } => Err(citadel_io::error!(
52            ErrorCode::MediaControlDecode,
53            "expected the peer's transport offer, got a media fragment"
54        )),
55    }
56}
