7/// Recommended upper bound on the plaintext bytes handed to the UDP channel per
8/// datagram across every transport. Measured floors: the P2P QUIC connection
9/// reports a 1162-byte datagram ceiling at spawn (=> 1066 B payload after the
10/// 64-byte HDP header + 32-byte AEAD at `SecurityLevel::Standard`); the raw
11/// hole-punched socket allows 1232 B datagrams (=> 1136 B). 1024 fits both
12/// with headroom for one extra AEAD layer.
13///
14/// The endpoint verifies the budget against the live transport's
15/// `OutboundUdpSender::max_payload_len()`; this constant is only a safe
16/// starting point for callers.
17pub const RECOMMENDED_UDP_PAYLOAD_BUDGET: usize = 1024;
18
19/// Recommended [`MediaConfig::max_fragment_payload`]:
20/// `RECOMMENDED_UDP_PAYLOAD_BUDGET - FRAGMENT_HEADER_LEN` = 1004, rounded down.
21pub const RECOMMENDED_FRAGMENT_PAYLOAD: usize = 1000;
22
23/// Transport tunables for a [`super::MediaEndpoint`]. Every field is mandatory
24/// (no `Default`); call [`MediaTransportConfig::validate`] before use.
25#[derive(Debug, Clone, Copy, PartialEq, Eq)]
26pub struct MediaTransportConfig {
27    /// Packetizer / reassembler / jitter-buffer settings.
28    pub media: MediaConfig,
29    /// Capacity of the outbound [`citadel_media::SendQueue`] in frames.
30    pub send_queue_frames: usize,
31    /// Maximum plaintext bytes per datagram handed to the unreliable sink.
32    /// Must be at least `media.max_fragment_payload + FRAGMENT_HEADER_LEN`.
33    pub udp_payload_budget: usize,
34    /// How long to wait for the session's UDP channel before falling back to
35    /// the ordered-reliable channel.
36    pub udp_wait: Duration,
37    /// Deadlines for agreeing on the transport in band with a peer that negotiates it (see
38    /// [`MediaAgreement::InBand`](super::MediaAgreement::InBand)).
39    pub negotiation: NegotiationPolicy,
40}
41
42impl MediaTransportConfig {
43    pub fn validate(&self) -> Result<(), NetworkError> {
44        self.media.validate().net()?;
45        self.negotiation.validate().map_err(|err| {
46            citadel_io::error!(
47                citadel_io::ErrorCode::MediaConfigInvalid,
48                format!("negotiation policy: {err:?}")
49            )
50        })?;
51        if self.send_queue_frames == 0 {
52            return Err(citadel_io::error!(
53                citadel_io::ErrorCode::MediaConfigInvalid,
54                "send_queue_frames must be > 0"
55            ));
56        }
57        let datagram = self
58            .media
59            .max_fragment_payload
60            .saturating_add(FRAGMENT_HEADER_LEN);
61        if datagram > self.udp_payload_budget {
62            return Err(citadel_io::error!(
63                citadel_io::ErrorCode::MediaConfigInvalid,
64                format!(
65                    "max_fragment_payload + {FRAGMENT_HEADER_LEN} = {datagram} exceeds udp_payload_budget {}",
66                    self.udp_payload_budget
67                )
68            ));
69        }
70        Ok(())
71    }
72
73    /// The largest datagram this config will ever emit.
74    pub const fn max_datagram_len(&self) -> usize {
75        self.media.max_fragment_payload + FRAGMENT_HEADER_LEN
76    }
77}
