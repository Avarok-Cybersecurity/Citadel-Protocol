1//! Deferred end-of-stream resolution: an `EndOfStream { track, frames_sent }`
2//! control message may outrace in-flight media, so it is held until the
3//! track's delivery reaches `frames_sent` or a jitter-depth deadline passes.
4use super::receiver::{queue_pop, MediaEvent};
5use citadel_media::{JitterBuffer, MediaInstant, MediaStats, TrackId};
6use std::collections::VecDeque;
7
8/// A pending end-of-stream announcement: surfaced once the track's delivery
9/// reaches `final_seq`, or force-closed (with a `Gap`) when `deadline` passes.
10#[derive(Debug, Clone, Copy)]
11struct EosRecord {
12    track: TrackId,
13    final_seq: u32,
14    deadline: MediaInstant,
15}
16
17/// Same wrapping sequence arithmetic as the jitter buffer.
18#[inline]
19const fn seq_diff(a: u32, b: u32) -> i32 {
20    a.wrapping_sub(b) as i32
21}
22
23#[derive(Debug)]
24pub(super) struct EosTracker {
25    records: Vec<EosRecord>,
26    /// From the validated `MediaConfig`; bounds how long an EOS may wait.
27    jitter_depth_micros: u64,
28}
29
30impl EosTracker {
31    pub(super) fn new(jitter_depth_micros: u64) -> Self {
32        Self {
33            records: Vec::new(),
34            jitter_depth_micros,
35        }
36    }
37
38    pub(super) fn record(&mut self, track: TrackId, frames_sent: u32, now: MediaInstant) {
39        self.records.push(EosRecord {
40            track,
41            final_seq: frames_sent,
42            deadline: now.saturating_add_micros(self.jitter_depth_micros),
43        });
44    }
45
46    pub(super) fn next_deadline(&self) -> impl Iterator<Item = MediaInstant> + '_ {
47        self.records.iter().map(|rec| rec.deadline)
48    }
49
50    /// Resolves pending records: a track whose delivery reached `final_seq`
51    /// emits `EndOfStream`. Once a record's deadline passes (or its transport
52    /// closed, `force`), the frames still buffered for its track are released
53    /// at once, skipping holes, and only the tail that never arrived is
54    /// reported as a `Gap` before `EndOfStream`. Returns whether any event was
55    /// queued.
56    pub(super) fn resolve(
57        &mut self,
58        now: MediaInstant,
59        force: bool,
60        jitter: &mut JitterBuffer,
61        ready: &mut VecDeque<MediaEvent>,
62        stats: &mut MediaStats,
63    ) -> bool {
64        let mut queued = false;
65        let mut i = 0;
66        while i < self.records.len() {
67            let rec = self.records[i];
68            let next = jitter.next_expected(rec.track);
69            let complete = match next {
70                Some(n) => seq_diff(n, rec.final_seq) >= 0,
71                None => rec.final_seq == 0,
72            };
73            if complete {
74                let _ = self.records.swap_remove(i);
75                ready.push_back(MediaEvent::EndOfStream(rec.track));
76                queued = true;
77                continue;
78            }
79            if force || now >= rec.deadline {
80                // Every frame of the track was sent before the EOS, which has
81                // now waited the jitter depth: no hole ahead of a buffered
82                // frame can still fill, so the frame is delivered, not lost.
83                if queue_pop(jitter.pop_track_now(rec.track, now), ready, stats) {
84                    queued = true;
85                    continue;
86                }
87                let _ = self.records.swap_remove(i);
88                let missing_from = next.unwrap_or(0);
89                let missing = seq_diff(rec.final_seq, missing_from);
90                if missing > 0 {
91                    stats.gaps_skipped += 1;
92                    stats.frames_missing += missing as u64;
93                    ready.push_back(MediaEvent::Gap {
94                        track: rec.track,
95                        missing_from,
96                        missing_to: rec.final_seq.wrapping_sub(1),
97                    });
98                }
99                ready.push_back(MediaEvent::EndOfStream(rec.track));
100                queued = true;
101                continue;
102            }
103            i += 1;
104        }
105        queued
106    }
107}
