1//! Deferred end-of-stream resolution: an `EndOfStream { track, frames_sent }`
2//! control message may outrace in-flight media, so it is held until the
3//! track's delivery reaches `frames_sent` or a jitter-depth deadline passes.
4use super::receiver::MediaEvent;
5use citadel_media::{MediaInstant, MediaStats, TrackId};
6use std::collections::VecDeque;
7
8/// A pending end-of-stream announcement: surfaced once the track's delivery
9/// reaches `final_seq`, or force-closed (with a `Gap`) when `deadline` passes.
10#[derive(Debug, Clone, Copy)]
11struct EosRecord {
12    track: TrackId,
13    final_seq: u32,
14    deadline: MediaInstant,
15}
16
17/// Same wrapping sequence arithmetic as the jitter buffer.
18#[inline]
19const fn seq_diff(a: u32, b: u32) -> i32 {
20    a.wrapping_sub(b) as i32
21}
22
23#[derive(Debug)]
24pub(super) struct EosTracker {
25    records: Vec<EosRecord>,
26    /// From the validated `MediaConfig`; bounds how long an EOS may wait.
27    jitter_depth_micros: u64,
28}
29
30impl EosTracker {
31    pub(super) fn new(jitter_depth_micros: u64) -> Self {
32        Self {
33            records: Vec::new(),
34            jitter_depth_micros,
35        }
36    }
37
38    pub(super) fn record(&mut self, track: TrackId, frames_sent: u32, now: MediaInstant) {
39        self.records.push(EosRecord {
40            track,
41            final_seq: frames_sent,
42            deadline: now.saturating_add_micros(self.jitter_depth_micros),
43        });
44    }
45
46    pub(super) fn next_deadline(&self) -> impl Iterator<Item = MediaInstant> + '_ {
47        self.records.iter().map(|rec| rec.deadline)
48    }
49
50    /// Resolves pending records: a track whose delivery reached `final_seq`
51    /// emits `EndOfStream`; one whose deadline passed (or whose transport
52    /// closed, `force`) emits a `Gap` over the lost tail first. Returns
53    /// whether any event was queued.
54    pub(super) fn resolve(
55        &mut self,
56        now: MediaInstant,
57        force: bool,
58        next_expected: impl Fn(TrackId) -> Option<u32>,
59        ready: &mut VecDeque<MediaEvent>,
60        stats: &mut MediaStats,
61    ) -> bool {
62        let mut queued = false;
63        let mut i = 0;
64        while i < self.records.len() {
65            let rec = self.records[i];
66            let next = next_expected(rec.track);
67            let complete = match next {
68                Some(n) => seq_diff(n, rec.final_seq) >= 0,
69                None => rec.final_seq == 0,
70            };
71            if complete {
72                let _ = self.records.swap_remove(i);
73                ready.push_back(MediaEvent::EndOfStream(rec.track));
74                queued = true;
75                continue;
76            }
77            if force || now >= rec.deadline {
78                let _ = self.records.swap_remove(i);
79                let missing_from = next.unwrap_or(0);
80                let missing = seq_diff(rec.final_seq, missing_from);
81                if missing > 0 {
82                    stats.gaps_skipped += 1;
83                    stats.frames_missing += missing as u64;
84                    ready.push_back(MediaEvent::Gap {
85                        track: rec.track,
86                        missing_from,
87                        missing_to: rec.final_seq.wrapping_sub(1),
88                    });
89                }
90                ready.push_back(MediaEvent::EndOfStream(rec.track));
91                queued = true;
92                continue;
93            }
94            i += 1;
95        }
96        queued
97    }
98}
