1//! A node's link mode is required configuration: [`NodeBuilder::build`] refuses a node built
2//! without one. There is no default mode (PCND): `Legacy` keeps today's behaviour, `Managed` turns
3//! on the RobustConnectionManager's, and the caller says which.
4
5use super::NodeBuilder;
6use crate::prelude::{LinkManagerMode, PlatformOps, Ratchet};
7use citadel_io::ErrorCode;
8
9impl<R: Ratchet, T: PlatformOps> NodeBuilder<R, T> {
10    /// How this node keeps its links: [`LinkManagerMode::Legacy`], or
11    /// [`LinkManagerMode::Managed`] with the clock its deadlines read and the `LinkPolicy` its
12    /// links are kept under (the reattach grace among them). Required.
13    pub fn with_link_manager(&mut self, mode: LinkManagerMode) -> &mut Self {
14        self.link_mode = Some(mode);
15        self
16    }
17}
18
19pub(super) fn required_link_mode(
20    mode: Option<LinkManagerMode>,
21) -> Result<LinkManagerMode, citadel_io::NetworkError> {
22    mode.ok_or_else(|| citadel_io::error!(ErrorCode::NodeLinkManagerMissing))
23}
24
25#[cfg(test)]
26mod tests {
27    use super::*;
28
29    #[test]
30    fn a_node_without_a_link_mode_is_refused() {
31        let err = required_link_mode(None).unwrap_err();
32        assert!(err.to_string().contains("link mode"), "{err}");
33        assert!(matches!(
34            required_link_mode(Some(LinkManagerMode::Legacy)),
35            Ok(LinkManagerMode::Legacy)
36        ));
37    }
38}
