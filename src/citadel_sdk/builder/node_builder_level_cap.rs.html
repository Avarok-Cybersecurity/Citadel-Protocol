1//! A node's maximum security level is required configuration: [`NodeBuilder::build`] refuses a
2//! node without one. There is no default maximum.
3
4use super::NodeBuilder;
5use crate::prelude::{PlatformOps, Ratchet};
6use citadel_io::ErrorCode;
7use citadel_types::crypto::MaxSecurityLevel;
8
9impl<R: Ratchet, T: PlatformOps> NodeBuilder<R, T> {
10    /// The most security layers this node builds a link with. Every level a peer, the server or
11    /// an unauthenticated registration names is checked against it before anything is built,
12    /// and one above it is refused. Required for every node.
13    pub fn with_max_security_level(&mut self, level: MaxSecurityLevel) -> &mut Self {
14        self.max_security_level = Some(level);
15        self
16    }
17}
18
19pub(super) fn required_max_security_level(
20    level: Option<MaxSecurityLevel>,
21) -> Result<MaxSecurityLevel, citadel_io::NetworkError> {
22    level.ok_or_else(|| citadel_io::error!(ErrorCode::NodeMaxSecurityLevelMissing))
23}

25#[cfg(all(test, not(target_family = "wasm")))]
26mod tests {
27    use crate::builder::node_builder::DefaultNodeBuilder;
28    use crate::prefabs::server::empty::EmptyKernel;
29    use crate::prelude::NodeType;
30    use crate::test_common::LOCALHOST_STUN_SERVERS;
31    use citadel_io::ErrorCode;
32    use citadel_types::crypto::{MaxSecurityLevel, SecurityLevel};
33
34    #[test]
35    fn a_node_without_a_maximum_security_level_is_refused() {
36        let mut builder = DefaultNodeBuilder::default();
37        let _ = builder
38            .with_link_manager(crate::prelude::LinkManagerMode::Legacy)
39            .with_node_type(NodeType::Peer)
40            .with_stun_servers(LOCALHOST_STUN_SERVERS);
41        let Err(err) = builder.build(EmptyKernel::default()) else {
42            panic!("a node without a maximum security level must be refused");
43        };
44        assert_eq!(
45            err.downcast_ref::<citadel_io::NetworkError>()
46                .expect("a typed error")
47                .code(),
48            ErrorCode::NodeMaxSecurityLevelMissing
49        );
50        let max = MaxSecurityLevel::new(SecurityLevel::Extreme).unwrap();
51        assert!(builder
52            .with_max_security_level(max)
53            .build(EmptyKernel::default())
54            .is_ok());
55    }
56}
