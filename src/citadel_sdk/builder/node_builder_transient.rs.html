1//! Whether a server allows transient accounts.
2//!
3//! A transient account is a session with its own CID that the server never persists and that has
4//! no password: the client proposes a fresh identifier, the server gives it a CID for the life of
5//! the connection, and nothing about it outlives the session. A client asks for one with
6//! `AuthenticationRequest::transient` or `ServerConnectionSettingsBuilder::transient`.
7//!
8//! The option is a server setting: [`NodeBuilder::build`] refuses it on a peer
9//! ([`ErrorCode::NodeTransientAccountsNotServer`]). It sets the one field the protocol reads,
10//! [`ServerMiscSettings::allow_transient_connections`], so there is a single source of truth.
11
12use super::NodeBuilder;
13use crate::prelude::{NodeType, PlatformOps, Ratchet, ServerMiscSettings};
14use citadel_io::{ErrorCode, NetworkError};
15
16impl<R: Ratchet, T: PlatformOps> NodeBuilder<R, T> {
17    /// Whether this server gives a client a transient account: a session with its own CID that
18    /// the server never persists and that has no password.
19    ///
20    /// **Without this call transient accounts are allowed.** That default is a deliberate product
21    /// decision approved by the project owner (2026-11-24), not an implicit fallback: a transient
22    /// account still has an identity (its CID) for as long as it is connected, and allowing it
23    /// keeps every existing server and client working as before.
24    ///
25    /// Passing `false` makes every account on this server a registered one. The server gives up
26    /// connections from clients that hold no account here (for example one that wants a single
27    /// short exchange before deciding to register), and gains the guarantee that every session
28    /// belongs to an account it stores. A refused client receives
29    /// [`ErrorCode::PasswordlessUnsupported`], distinct from every other failure; registered
30    /// accounts are unaffected.
31    ///
32    /// Refused on a peer ([`ErrorCode::NodeTransientAccountsNotServer`]). With
33    /// [`Self::with_server_misc_settings`] as well: `false` here wins whatever the settings hold
34    /// (their field cannot say whether it was chosen or defaulted), and `true` here with settings
35    /// that say `false` is a contradiction `build` refuses
36    /// ([`ErrorCode::NodeTransientAccountsConflict`]).
37    pub fn with_transient_accounts(&mut self, allowed: bool) -> &mut Self {
38        self.transient_accounts = Some(allowed);
39        self
40    }
41}
42
43pub(super) fn check_transient_accounts(
44    node_type: Option<&NodeType>,
45    transient_accounts: Option<bool>,
46) -> Result<(), NetworkError> {
47    let is_server = matches!(node_type, Some(NodeType::Server(_)));
48    match (is_server, transient_accounts) {
49        (false, Some(_)) => Err(citadel_io::error!(
50            ErrorCode::NodeTransientAccountsNotServer
51        )),
52        _ => Ok(()),
53    }
54}
55
56/// The server misc settings with the option folded into their one transient field.
57pub(super) fn with_transient_accounts(
58    misc: Option<ServerMiscSettings>,
59    transient_accounts: Option<bool>,
60) -> Result<Option<ServerMiscSettings>, NetworkError> {
61    match (transient_accounts, misc) {
62        (None, misc) => Ok(misc),
63        (Some(true), Some(misc)) if !misc.allow_transient_connections => {
64            Err(citadel_io::error!(ErrorCode::NodeTransientAccountsConflict))
65        }
66        (Some(allowed), misc) => Ok(Some(ServerMiscSettings {
67            allow_transient_connections: allowed,
68            ..misc.unwrap_or_default()
69        })),
70    }
71}

73#[cfg(all(test, not(target_family = "wasm")))]
74mod tests {
75    use super::with_transient_accounts;
76    use crate::builder::node_builder::DefaultNodeBuilder;
77    use crate::prefabs::server::empty::EmptyKernel;
78    use crate::prelude::{NodeType, ServerMiscSettings};
79    use crate::test_common::LOCALHOST_STUN_SERVERS;
80    use citadel_io::ErrorCode;
81
82    fn misc(allowed: bool) -> Option<ServerMiscSettings> {
83        Some(ServerMiscSettings {
84            allow_transient_connections: allowed,
85            ..Default::default()
86        })
87    }
88
89    fn resolved(misc: Option<ServerMiscSettings>, option: Option<bool>) -> Option<bool> {
90        with_transient_accounts(misc, option)
91            .expect("no contradiction")
92            .map(|misc| misc.allow_transient_connections)
93    }
94
95    #[test]
96    fn the_option_sets_the_one_field_the_protocol_reads() {
97        assert_eq!(resolved(None, Some(false)), Some(false));
98        assert_eq!(resolved(None, Some(true)), Some(true));
99    }
100
101    #[test]
102    fn without_the_option_the_settings_are_untouched() {
103        assert_eq!(resolved(None, None), None);
104        assert_eq!(resolved(misc(false), None), Some(false));
105        assert_eq!(resolved(misc(true), None), Some(true));
106    }
107
108    #[test]
109    fn disallowing_wins_over_the_settings() {
110        assert_eq!(resolved(misc(true), Some(false)), Some(false));
111        assert_eq!(resolved(misc(false), Some(false)), Some(false));
112        assert_eq!(resolved(misc(true), Some(true)), Some(true));
113    }
114
115    #[test]
116    fn allowing_against_settings_that_disallow_is_refused() {
117        let Err(err) = with_transient_accounts(misc(false), Some(true)) else {
118            panic!("a contradiction was resolved silently");
119        };
120        assert_eq!(err.code(), ErrorCode::NodeTransientAccountsConflict);
121    }
122
123    #[test]
124    fn the_rest_of_the_settings_survive_the_option() {
125        let custom = ServerMiscSettings {
126            allow_transient_connections: true,
127            credential_requirements: citadel_user::credentials::CredentialRequirements {
128                min_password_length: 13,
129                ..Default::default()
130            },
131            ..Default::default()
132        };
133        let folded = with_transient_accounts(Some(custom), Some(false))
134            .unwrap()
135            .unwrap();
136        assert_eq!(folded.credential_requirements.min_password_length, 13);
137    }
138
139    fn build_error(node_type: NodeType, option: Option<bool>) -> Option<ErrorCode> {
140        let mut builder = DefaultNodeBuilder::default();
141        let _ = builder
142            .with_node_type(node_type)
143            .with_link_manager(crate::prelude::LinkManagerMode::Legacy)
144            .with_max_security_level(crate::test_common::localhost_max_security_level())
145            .with_stun_servers(LOCALHOST_STUN_SERVERS);
146        if node_type.is_server() {
147            let _ = builder
148                .with_path_policy(crate::test_common::localhost_path_policy())
149                .with_server_limits(crate::test_common::localhost_server_limits());
150        }
151        if let Some(allowed) = option {
152            let _ = builder.with_transient_accounts(allowed);
153        }
154        match builder.build(EmptyKernel::default()) {
155            Ok(_) => None,
156            Err(err) => Some(
157                err.downcast_ref::<citadel_io::NetworkError>()
158                    .expect("a typed error")
159                    .code(),
160            ),
161        }
162    }
163
164    #[test]
165    fn a_peer_given_the_option_is_refused() {
166        for allowed in [true, false] {
167            assert_eq!(
168                build_error(NodeType::Peer, Some(allowed)),
169                Some(ErrorCode::NodeTransientAccountsNotServer)
170            );
171        }
172        assert_eq!(build_error(NodeType::Peer, None), None);
173    }
174
175    #[test]
176    fn a_server_takes_the_option_either_way() {
177        let server = NodeType::server("127.0.0.1:0").unwrap();
178        for option in [Some(true), Some(false), None] {
179            assert_eq!(build_error(server, option), None);
180        }
181    }
182
183    #[test]
184    fn a_contradiction_fails_the_build() {
185        let mut builder = DefaultNodeBuilder::default();
186        let _ = builder
187            .with_node_type(NodeType::server("127.0.0.1:0").unwrap())
188            .with_link_manager(crate::prelude::LinkManagerMode::Legacy)
189            .with_max_security_level(crate::test_common::localhost_max_security_level())
190            .with_stun_servers(LOCALHOST_STUN_SERVERS)
191            .with_path_policy(crate::test_common::localhost_path_policy())
192            .with_server_limits(crate::test_common::localhost_server_limits())
193            .with_server_misc_settings(misc(false).unwrap())
194            .with_transient_accounts(true);
195        let Err(err) = builder.build(EmptyKernel::default()) else {
196            panic!("a contradiction was built");
197        };
198        let code = err
199            .downcast_ref::<citadel_io::NetworkError>()
200            .expect("a typed error")
201            .code();
202        assert_eq!(code, ErrorCode::NodeTransientAccountsConflict);
203    }
