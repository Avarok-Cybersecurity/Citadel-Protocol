1//! A server's limits are required configuration: [`NodeBuilder::build`] refuses a server node
2//! without them, and a peer node with them. There are no default limits.
3
4use super::NodeBuilder;
5use crate::prelude::{
6    GroupLogLimits, GroupMailLimits, NodeType, PlatformOps, Ratchet, ServerLimits,
7    ServerOnlySessionInitSettings,
8};
9use citadel_io::ErrorCode;
10
11impl<R: Ratchet, T: PlatformOps> NodeBuilder<R, T> {
12    /// How much this server holds for connections that have not signed in, how it takes legacy
13    /// registrations, and how much its streams may queue (see [`ServerLimits`]). Required for
14    /// [`NodeType::Server`]; refused on a peer.
15    pub fn with_server_limits(&mut self, limits: ServerLimits) -> &mut Self {
16        self.server_limits = Some(limits);
17        self
18    }
19
20    /// The most this server holds for one member of one group while it is away (see
21    /// [`GroupMailLimits`]). Without this call a server holds [`GroupMailLimits::default`];
22    /// refused on a peer.
23    pub fn with_group_mail_limits(&mut self, limits: GroupMailLimits) -> &mut Self {
24        self.group_mail_limits = Some(limits);
25        self
26    }
27
28    /// The most this server keeps of one replay group's log (see [`GroupLogLimits`]); a group
29    /// asks for less, never more. Without this call the server keeps no log, and refuses a group
30    /// replay (`GroupReplayRefusal::NotKept`); refused on a peer.
31    pub fn with_group_log_limits(&mut self, limits: GroupLogLimits) -> &mut Self {
32        self.group_log_limits = Some(limits);
33        self
34    }
35}
36
37pub(super) fn check_server_limits(
38    node_type: Option<&NodeType>,
39    limits: Option<&ServerLimits>,
40    group_mail_limits: bool,
41) -> Result<(), citadel_io::NetworkError> {
42    let is_server = matches!(node_type, Some(NodeType::Server(_)));
43    match (is_server, limits.is_some() || group_mail_limits) {
44        (true, _) if limits.is_none() => {
45            Err(citadel_io::error!(ErrorCode::NodeServerLimitsMissing))
46        }
47        (false, true) => Err(citadel_io::error!(ErrorCode::NodeServerLimitsNotServer)),
48        _ => Ok(()),
49    }
50}
51
52/// The server-only session settings with the limits folded in. A peer has neither.
53pub(super) fn with_limits(
54    settings: Option<ServerOnlySessionInitSettings>,
55    limits: Option<ServerLimits>,
56    group_mail: Option<GroupMailLimits>,
57    group_log: Option<GroupLogLimits>,
58) -> Option<ServerOnlySessionInitSettings> {
59    match limits {
60        Some(limits) => {
61            let mut settings = settings.unwrap_or_default();
62            settings.limits = Some(limits);
63            settings.group_mail = group_mail.unwrap_or_default();
64            settings.group_log = group_log;
65            Some(settings)
66        }
67        None => settings,
68    }
69}

72mod tests {
73    use crate::builder::node_builder::DefaultNodeBuilder;
74    use crate::prefabs::server::empty::EmptyKernel;
75    use crate::prelude::NodeType;
76    use crate::test_common::{

79    use citadel_io::ErrorCode;

81    fn build_error(node_type: NodeType, with_limits: bool) -> Option<ErrorCode> {
82        let mut builder = DefaultNodeBuilder::default();
83        let _ = builder
84            .with_node_type(node_type)
85            .with_link_manager(crate::prelude::LinkManagerMode::Legacy)
86            .with_max_security_level(crate::test_common::localhost_max_security_level())
87            .with_stun_servers(LOCALHOST_STUN_SERVERS);
88        if node_type.is_server() {
89            let _ = builder.with_path_policy(localhost_path_policy());
90        }
91        if with_limits {
92            let _ = builder.with_server_limits(localhost_server_limits());
93        }
94        match builder.build(EmptyKernel::default()) {
95            Ok(_) => None,
96            Err(err) => Some(
97                err.downcast_ref::<citadel_io::NetworkError>()
98                    .expect("a typed error")
99                    .code(),
100            ),
101        }
102    }
103
104    #[test]
105    fn a_server_without_limits_is_refused() {
106        let server = NodeType::server("127.0.0.1:0").unwrap();
107        assert_eq!(
108            build_error(server, false),
109            Some(ErrorCode::NodeServerLimitsMissing)
110        );
111        assert_eq!(build_error(server, true), None);
112    }
113
114    #[test]
115    fn a_peer_with_limits_is_refused() {
116        assert_eq!(
117            build_error(NodeType::Peer, true),
118            Some(ErrorCode::NodeServerLimitsNotServer)
119        );
120        assert_eq!(build_error(NodeType::Peer, false), None);
121    }
122
123    #[test]
124    fn group_mail_limits_reach_the_server_and_default_without_the_call() {
125        use super::with_limits;
126        use crate::prelude::GroupMailLimits;
127        let custom = GroupMailLimits {
128            max_age: std::time::Duration::from_secs(60),
129            max_messages: std::num::NonZeroUsize::new(3).unwrap(),
130            max_bytes: std::num::NonZeroUsize::new(4096).unwrap(),
131        };
132        let given = with_limits(None, Some(localhost_server_limits()), Some(custom), None);
133        assert_eq!(given.unwrap().group_mail, custom);
134        let defaulted = with_limits(None, Some(localhost_server_limits()), None, None);
135        assert_eq!(defaulted.unwrap().group_mail, GroupMailLimits::default());
136    }
137
138    #[test]
139    fn group_mail_limits_on_a_peer_are_refused() {
140        let mut builder = DefaultNodeBuilder::default();
141        let _ = builder
142            .with_node_type(NodeType::Peer)
143            .with_link_manager(crate::prelude::LinkManagerMode::Legacy)
144            .with_max_security_level(crate::test_common::localhost_max_security_level())
145            .with_stun_servers(LOCALHOST_STUN_SERVERS)
146            .with_group_mail_limits(crate::prelude::GroupMailLimits::default());
147        let Err(err) = builder.build(EmptyKernel::default()) else {
148            panic!("a peer with group mail limits was built");
149        };
150        let code = err
151            .downcast_ref::<citadel_io::NetworkError>()
152            .expect("a typed error")
153            .code();
154        assert_eq!(code, ErrorCode::NodeServerLimitsNotServer);
155    }
