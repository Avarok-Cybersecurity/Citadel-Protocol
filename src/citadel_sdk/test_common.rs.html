1//! Testing Utilities for Citadel Protocol
2//!
3//! This module provides common utilities and helpers for testing Citadel Protocol
4//! components. It includes functions for setting up test servers, managing test
5//! barriers, and performing common assertions.
6//!
7//! # Features
8//! - Test server creation and configuration
9//! - Synchronization barriers for multi-peer tests
10//! - UDP mode testing utilities
11//! - P2P connection testing helpers
12//! - Local test peer management
13//!
14//! # Example
15//! ```rust
16//! use citadel_sdk::test_common::*;
17//! use citadel_sdk::prelude::*;
18//!
19//! async fn test_server() {
20//!     // Create a test server with default settings
21//!     let (server_future, addr) = server_info::<StackedRatchet>();
22//!
23//!     // Run server and handle connections
24//!     server_future.await.expect("Server failed to start");
25//! }
26//! ```
27//!
28//! # Important Notes
29//! - Most functionality requires "localhost-testing" feature
30//! - Test barriers help coordinate multi-peer tests
31//! - UDP assertions verify connection modes
32//! - P2P assertions validate peer connections
33//!
34//! # Related Components
35//! - [`DefaultNodeBuilder`]: Server configuration builder
36//! - [`EmptyKernel`]: Basic test server kernel
37//! - [`UdpMode`]: UDP connection configuration
38//! - [`PeerConnectSuccess`]: Peer connection validation
39//!

82/// The path policy of servers under test: the owner's order (Direct, PeerRelay, Turn,
83/// ServerRelay), every path enabled for every class. A server requires a policy and there is no
84/// default; this one keeps the attempt timeouts the campaign used before policies existed (30 s
85/// per path), so tests written against that schedule still hold.
86pub fn localhost_path_policy() -> PathPolicy {
87    let rules = PathKind::ALL
88        .into_iter()
89        .map(|kind| PathRule {
90            kind,
91            enabled: true,
92            attempt_timeout: Duration::from_secs(30),
93            cost_weight: 1,
94            classes: PathClasses::ALL,
95        })
96        .collect();
97    PathPolicy::new(rules, PathClasses::NONE).expect("the test policy is valid")
98}

100/// The limits of servers under test. Every client of a test dials from the one loopback address
101/// and many tests start dozens of clients at once, so the per-address limit is the total; both
102/// are far above any suite's concurrency. Legacy registration is taken (suites exercise it),
103/// with as many concurrent hashes as the largest suite registers at once. The queue limits are
104/// the per-stream bound clients use (`outbound_budget::MAX_QUEUED_OUTBOUND_BYTES`, 256 MiB) and
105/// four of them in all (1 GiB), which no suite's transfers approach. The total is a `const` so a
106/// value past a 32-bit `usize` (wasm32) fails to compile under `cargo check` too.
107pub fn localhost_server_limits() -> ServerLimits {
108    const CONNECTIONS: usize = 1024;
109    const QUEUED_PER_STREAM: usize = 256 * 1024 * 1024;
110    const QUEUED_TOTAL: usize = 4 * QUEUED_PER_STREAM;
111    let nonzero = |n: usize| std::num::NonZeroUsize::new(n).expect("a positive test limit");
112    ServerLimits {
113        provisional_per_ip: nonzero(CONNECTIONS),
114        provisional_total: nonzero(CONNECTIONS),
115        legacy_registration: LegacyRegistration::Accepted {
116            concurrent_hashes: nonzero(64),
117        },
118        outbound_queued_per_stream: nonzero(QUEUED_PER_STREAM),
119        outbound_queued_total: nonzero(QUEUED_TOTAL),
120    }
121}

123/// The link policy of managed nodes under test. `LinkPolicy` has no defaults, so every bound is
124/// explicit. The values are the campaign's own (2 s first retry, 60 s cap, 5 s rearm floor), so
125/// the two modes are timed alike, and the reattach grace is the owner's 5 minutes (Q2).
126pub fn localhost_link_policy() -> LinkPolicy {
127    let s = Duration::from_secs;
128    LinkPolicy {
129        heal: HealPolicy {
130            backoff: Backoff {
131                first_wait: s(2),
132                factor: 2,
133                max_wait: s(60),
134            },
135            max_attempts: Some(6),
136            give_up_after: None,
137        },
138        reattach_grace: s(300),
139        flap: FlapPolicy {
140            window: s(60),
141            max_transitions: 5,
142            hold: s(20),
143        },
144        direct_rendezvous_budget: s(5),
145        rendezvous_deadline: s(30),
146        seal_deadline: s(30),
147        probe_interval: None,
148        settings_deadline: s(30),
149        loser_redial_wait: s(2),
150        rearm: Backoff {
151            first_wait: s(5),
152            factor: 2,
153            max_wait: s(300),
154        },
155    }

214    fn server_test_node_inner<'a, K: NetKernel<R> + 'a, R: Ratchet>(
215        kernel: K,
216        opts: impl FnOnce(&mut NodeBuilder<R>),
217        enable_websocket: bool,
218    ) -> ((NodeFuture<'a, K>, SocketAddr), SocketAddr) {
219        let mut builder = node_builder_for::<R>();
220        let _ = builder.with_link_manager(LinkManagerMode::Legacy);
221        // A `ServerMode::P2P` override rebinds this explicit port in TCP and UDP, so the port is
222        // drawn by the server's own rule for a pair (a port drawn for TCP alone can sit in a range
223        // reserved for UDP), and the UDP half is released for that rebind.
224        let (tcp_listener, _udp) =
225            citadel_proto::re_imports::server_socket_pair(SocketAddr::from(([127, 0, 0, 1], 0)))
226                .expect("Failed to draw a loopback port free in both TCP and UDP");
227        let bind_addr = tcp_listener.local_addr().unwrap();
228        let builder = builder
229            .with_node_type(NodeType::Server(bind_addr))
230            .with_stun_servers(LOCALHOST_STUN_SERVERS)
231            .with_max_security_level(crate::test_common::localhost_max_security_level())
232            .with_path_policy(localhost_path_policy())
233            .with_server_limits(localhost_server_limits())
234            .with_underlying_protocol(ServerMode::OrderedReliable(
235                NativeOrderedReliableConfig::from_tokio_listener(tcp_listener).unwrap(),
236            ));
237
238        let ws_addr = if enable_websocket {
239            let ws_listener = citadel_wire::socket_helpers::get_tcp_listener("127.0.0.1:0")
240                .expect("Failed to create WebSocket TCP listener");
241            let ws_addr = ws_listener.local_addr().unwrap();
242            let _ = builder.with_bound_websocket_listener(ws_listener.into_std().unwrap());
243            ws_addr
244        } else {
245            SocketAddr::from(([0, 0, 0, 0], 0))
246        };
247
248        (opts)(builder);
249
250        ((builder.build::<K>(kernel).unwrap(), bind_addr), ws_addr)
251    }

308pub async fn wait_for_peers() {
309    let barrier = { TEST_BARRIER.lock().clone() };
310    assert!(*DEADLOCK_INIT, "Deadlock detector not initialized");
311    if let Some(test_barrier) = barrier {
312        // Wait for all peers to reach this point in the code
313        test_barrier.wait().await;
314    }
315}

348#[cfg(feature = "localhost-testing")]
349impl TestBarrier {
350    pub fn setup(count: usize) {
351        assert!(TEST_BARRIER.lock().replace(Self::new(count)).is_none(), "TestBarrier already set up. Make sure to run tests in separate program spaces to ensure that the barrier is not shared across tests. E.g., run with `cargo nextest run` instead of `cargo test`");
352    }
353    #[allow(dead_code)]
354    pub(crate) fn new(count: usize) -> Self {
355        Self {
356            inner: std::sync::Arc::new(citadel_io::tokio::sync::Barrier::new(count)),
357            count,
358        }
359    }
360    pub async fn wait(&self) {
361        let _ = self.inner.wait().await;
362    }
363
364    /// Unconditionally (re)installs the global test barrier. Unlike [`Self::setup`], which panics if
365    /// a barrier is already present, this replaces any existing one. Intended for benches that run
366    /// several peer-count configs back-to-back in a single process (each config needs a fresh barrier
367    /// sized to that config's node count); not for the nextest-per-process test flow.
368    pub fn reset(count: usize) {
369        *TEST_BARRIER.lock() = Some(Self::new(count));
370    }
371}

387// `DEADLOCK_INIT` is always `true`; the background deadlock-detector thread only spawns when the
388// opt-in `deadlock-detection` feature is on (it requires `citadel_io::deadlock`, which only exists
389// then). This keeps `wait_for_peers`'s `assert!(*DEADLOCK_INIT)` valid in both configurations while
390// letting benches run without the process-global lock tracker.
391#[cfg(feature = "localhost-testing")]
392lazy_static::lazy_static! {
393    static ref DEADLOCK_INIT: bool = {
394        #[cfg(feature = "deadlock-detection")]
395        {
396            let _ = std::thread::spawn(move || {
397                log::trace!(target: "citadel", "Executing deadlock detector ...");
398                use std::time::Duration;
399                use citadel_io::deadlock;
400                loop {
401                    std::thread::sleep(Duration::from_secs(5));
402                    let deadlocks = deadlock::check_deadlock();
403                    if deadlocks.is_empty() {
404                        log::trace!(target: "citadel", "No deadlocks detected");
405                        continue;
406                    }
407
408                    log::error!(target: "citadel", "{} deadlocks detected", deadlocks.len());
409                    for (i, threads) in deadlocks.iter().enumerate() {
410                        log::error!(target: "citadel", "Deadlock #{}", i);
411                        for t in threads {
412                            log::error!(target: "citadel", "Thread Id {:#?}", t.thread_id());
413                            log::error!(target: "citadel", "{:#?}", t.backtrace());
414                        }
415                    }
416                }
417            });
418        }
419
420        true
421    };
422}

424/// Releases a split UDP channel without racing the peer: both sides rendezvous on the test
425/// barrier first, so the `DisconnectUDP` fired by dropping the receive half only ever tears down
426/// an idle subsystem. Use this instead of `std::mem::forget` in barrier-synchronised tests.
427#[cfg(feature = "localhost-testing")]
428pub async fn finish_udp_channel<R: Ratchet>(
429    tx: citadel_proto::prelude::OutboundUdpSender,
430    rx: citadel_proto::prelude::PeerChannelRecvHalf<R>,
431) {
432    wait_for_peers().await;
433    drop((tx, rx));
434}

440#[allow(dead_code, clippy::cognitive_complexity)]
441pub async fn udp_mode_assertions<R: Ratchet>(
442    udp_mode: UdpMode,
443    udp_channel_rx_opt: Option<citadel_io::tokio::sync::oneshot::Receiver<UdpChannel<R>>>,
444) {
445    use futures::StreamExt;
446    citadel_logging::info!(target: "citadel", "Inside UDP mode assertions ...");
447    match udp_mode {
448        UdpMode::Enabled => {
449            citadel_logging::info!(target: "citadel", "Inside UDP mode assertions AB1 ...");
450            assert!(udp_channel_rx_opt.is_some());
451            citadel_logging::info!(target: "citadel", "Inside UDP mode assertions AB1.5 ...");
452            let chan = udp_channel_rx_opt.unwrap().await.unwrap();
453            citadel_logging::info!(target: "citadel", "Inside UDP mode assertions AB2 ...");
454            let (tx, mut rx) = chan.split();
455
456            // UDP does not promise delivery, and this used to assert that it
457            // does: two datagrams sent, two receives awaited, both unbounded.
458            // One dropped datagram blocked `rx.next()` for the rest of the test
459            // and died at the 180s timeout with no indication of where.
460            //
461            // Every connected PAIR runs this, so the exposure scales with the
462            // peer count — which is exactly the observed pattern:
463            // `test_peer_to_peer_file_transfer::case_2` (3 peers, 3 pairs)
464            // timed out where `case_1` (2 peers, 1 pair) passed in seconds.
465            //
466            // Resending until the echo arrives tests what was meant — that the
467            // channel carries datagrams — without asserting a guarantee UDP has
468            // never made. The bound turns a silent 180s hang into a failure that
469            // says what did not happen.
470            const RESEND_EVERY: Duration = Duration::from_millis(200);
471            const RESENDS_BEFORE_GIVING_UP: usize = 150; // 30s
472            for exchange in ["first", "second"] {
473                let mut delivered = false;
474                for _ in 0..RESENDS_BEFORE_GIVING_UP {
475                    tx.unbounded_send(b"Hello, world!" as &[u8]).unwrap();
476                    match citadel_io::time::timeout(RESEND_EVERY, rx.next()).await {
477                        Ok(Some(message)) => {
478                            assert_eq!(message.as_ref(), b"Hello, world!");
479                            delivered = true;
480                            break;
481                        }
482                        Ok(None) => panic!("the UDP channel closed during the {exchange} exchange"),
483                        Err(_) => continue, // dropped; send it again
484                    }
485                }
486                assert!(
487                    delivered,
488                    "no UDP datagram came back in 30s during the {exchange} exchange",
489                );
490                if exchange == "first" {
491                    citadel_logging::info!(target: "citadel", "Inside UDP mode assertions AB2.5 ...");
492                }
493            }
494            // Keep SENDING while waiting, not just waiting.
495            //
496            // The exchange is mutual: each side sends and each side expects to
497            // receive. A peer that completes first used to stop sending and
498            // sleep, which strands a peer whose datagram was lost — it resends
499            // into somebody who will never answer. That is why resend-only
500            // recovered the 2-peer case and not the 3-peer one, where a peer
501            // holds two connections and finishes one before the other.
502            for _ in 0..15 {
503                let _ = tx.unbounded_send(b"Hello, world!" as &[u8]);
504                citadel_io::time::sleep(Duration::from_millis(200)).await;
505            }
506            //wait_for_peers().await;
507            std::mem::forget((tx, rx)); // do not run destructor to not trigger premature
508        }
509
510        UdpMode::Disabled => {
511            citadel_logging::info!(target: "citadel", "Inside UDP mode assertions AB0-null ...");
512            assert!(udp_channel_rx_opt.is_none());
513        }
514    }
515
516    log::info!(target: "citadel", "Done w/ UDP mode assertions");
517}

521pub async fn p2p_assertions<R: Ratchet>(session_cid: u64, conn_success: &PeerConnectSuccess<R>) {
522    log::info!(target: "citadel", "Inside p2p assertions ...");
523    let peer_cid = conn_success.channel.get_peer_cid();
524
525    assert_eq!(session_cid, conn_success.channel.get_session_cid());
526    assert_ne!(session_cid, peer_cid);
527    assert!(conn_success
528        .remote
529        .inner
530        .account_manager()
531        .get_persistence_handler()
532        .hyperlan_peer_exists(session_cid, peer_cid)
533        .await
534        .unwrap());
535
536    log::info!(target: "citadel", "Done w/ p2p mode assertions");
537}
