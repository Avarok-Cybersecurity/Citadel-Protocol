1//! Driving a group-creation subscription to its terminal event.
2//!
3//! Split out of `remote_ext.rs` so the refusal paths are testable without a
4//! running node: the loop lived inside `create_group_with_options`, whose only
5//! reachable failure is a server that already holds 256 groups for the owner —
6//! 257 protocol round trips to stage, which is not a test anyone should have to
7//! run.
8//!
9//! The loop matched `GroupChannelCreated` and nothing else. It did not call
10//! `into_result()` — alone among its neighbours — so `SignalError`,
11//! `OutboundRequestRejected` and `InternalServerError` were discarded, and it
12//! discarded the server's own answer to a Create it could not perform:
13//! `CreateResponse { key: None }`, delivered on this very ticket. The
14//! subscription's stream ends only when its receiver drops, and the receiver is
15//! what the waiting caller holds, so every one of those meant a caller parked
16//! for the life of the process.
17
18use crate::prelude::*;
19use futures::{Stream, StreamExt};
20
21/// How a group creation ended.
22pub(crate) enum GroupCreation {
23    /// The channel opened.
24    Created(GroupChannel),
25    /// `CreateResponse { key: None }` — the server would not create it.
26    /// `create_new_message_group` returns `None` when the owner already holds
27    /// 256 groups, or when their `message_groups` entry is missing.
28    Refused,
29    /// The kernel dropped the subscription without a terminal event.
30    Ended,
31}
32
33/// Drive `events` to the creation's terminal event.
34///
35/// Errors carried by the stream are returned rather than skipped, which is what
36/// `into_result()` buys and what this loop was missing.
37pub(crate) async fn await_group_creation<R: Ratchet, S>(
38    events: &mut S,
39) -> Result<GroupCreation, NetworkError>
40where
41    S: Stream<Item = NodeResult<R>> + Unpin,
42{
43    while let Some(evt) = events.next().await {
44        match evt.into_result()? {
45            NodeResult::GroupChannelCreated(GroupChannelCreated { channel, .. }) => {
46                return Ok(GroupCreation::Created(channel))
47            }
48            NodeResult::GroupEvent(GroupEvent {
49                event: GroupBroadcast::CreateResponse { key: None },
50                ..
51            }) => return Ok(GroupCreation::Refused),
52            _ => {}
53        }
54    }
55    Ok(GroupCreation::Ended)
56}
57
58#[cfg(test)]
59mod tests {
60    use super::*;
61    use citadel_io::tokio;
62
63    type R = StackedRatchet;
64
65    fn key() -> MessageGroupKey {
66        MessageGroupKey::new(1, 1)
67    }
68
69    fn group_event(event: GroupBroadcast) -> NodeResult<R> {
70        NodeResult::GroupEvent(GroupEvent {
71            session_cid: 1,
72            ticket: Ticket(1),
73            event,
74        })
75    }
76
77    /// A non-terminal event the wait must step over rather than take as an
78    /// answer: the outbound request form, not a response to it.
79    fn non_terminal() -> NodeResult<R> {
80        group_event(GroupBroadcast::Create {
81            initial_invitees: vec![],
82            options: Default::default(),
83        })
84    }
85
86    /// The event the loop used to drop on the floor.
87    #[citadel_io::tokio::test]
88    async fn a_refusal_is_terminal() {
89        let mut events = futures::stream::iter(vec![
90            non_terminal(),
91            group_event(GroupBroadcast::CreateResponse { key: None }),
92        ]);
93        assert!(matches!(
94            await_group_creation::<R, _>(&mut events).await.unwrap(),
95            GroupCreation::Refused
96        ));
97    }
98
99    /// A SUCCESSFUL create also answers with `CreateResponse`, carrying a key.
100    /// Treating that as the refusal would break every group creation, so the
101    /// match is on `key: None` and this is the test that says so.
102    #[citadel_io::tokio::test]
103    async fn a_successful_create_response_is_not_a_refusal() {
104        let mut events = futures::stream::iter(vec![group_event(GroupBroadcast::CreateResponse {
105            key: Some(key()),
106        })]);
107        assert!(matches!(
108            await_group_creation::<R, _>(&mut events).await.unwrap(),
109            GroupCreation::Ended
110        ));
111    }
112
113    /// `into_result()` is the part the loop was missing. An error on the stream
114    /// must come back as an error, not be stepped over on the way to a terminal
115    /// event that will never arrive.
116    #[citadel_io::tokio::test]
117    async fn an_error_on_the_stream_is_returned() {
118        let mut events =
119            futures::stream::iter(vec![NodeResult::InternalServerError(InternalServerError {
120                ticket_opt: Some(Ticket(1)),
121                cid_opt: Some(1),
122                message: "the node refused the request".to_string(),
123            })]);
124        let Err(err) = await_group_creation::<R, _>(&mut events).await else {
125            panic!("an InternalServerError must not be skipped");
126        };
127        assert!(
128            err.into_string().contains("the node refused the request"),
129            "the node's reason must survive to the caller",
130        );
131    }
132
133    /// A subscription the kernel drops without answering is a failure, not a
134    /// hang and not a success.
135    #[citadel_io::tokio::test]
136    async fn an_ended_subscription_is_not_an_answer() {
137        let mut events = futures::stream::iter(Vec::<NodeResult<R>>::new());
138        assert!(matches!(
139            await_group_creation::<R, _>(&mut events).await.unwrap(),
140            GroupCreation::Ended
141        ));
142    }
143}
