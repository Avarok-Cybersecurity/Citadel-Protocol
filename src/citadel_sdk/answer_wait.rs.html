1//! Waiting, with a deadline, for the event that answers a request.
2//!
3//! A request's subscription carries only events for that request, but it can stay open and
4//! never carry the answer: a peer or server that never replies. Every wait is therefore bounded;
5//! an unanswered request is reported, which a caller can act on, instead of parking it for ever.
6
7use citadel_proto::prelude::{NetworkError, NodeResult, Ratchet};
8use futures::{Stream, StreamExt};
9use std::time::Duration;
10
11/// How long a disconnect or deregistration waits for the server's confirmation. The server
12/// answers these itself, so a live one answers at once.
13pub(crate) const SERVER_ANSWER_TIMEOUT: Duration = Duration::from_secs(30);
14
15/// Waits up to `within` for the event `answer` recognises, skipping events it does not. An error
16/// event on the subscription is the request's failure and is returned as such. `unanswered`
17/// makes the error for a deadline passed or a subscription ended with no answer.
18pub(crate) async fn await_answer<R, S, T>(
19    subscription: &mut S,
20    within: Duration,
21    what: &str,
22    unanswered: impl Fn() -> NetworkError,
23    mut answer: impl FnMut(NodeResult<R>) -> Option<Result<T, NetworkError>>,
24) -> Result<T, NetworkError>
25where
26    R: Ratchet,
27    S: Stream<Item = NodeResult<R>> + Unpin,
28{
29    let deadline = citadel_io::time::Instant::now() + within;
30    // What the subscription did carry, for the occasion when it does not carry the answer: a
31    // non-zero count says it was alive and the answer was never emitted or went elsewhere; zero
32    // says it heard nothing at all, which points at the session rather than the routing.
33    let mut carried = 0usize;
34    loop {
35        let remaining = deadline.saturating_duration_since(citadel_io::time::Instant::now());
36        match citadel_io::time::timeout(remaining, subscription.next()).await {
37            Ok(Some(event)) => {
38                carried += 1;
39                if let Some(outcome) = answer(event.into_result()?) {
40                    return outcome;
41                }
42            }
43            Ok(None) => {
44                log::warn!(target: "citadel", "[answer-wait] the subscription for {what} closed unanswered; it carried {carried} other event(s)");
45                return Err(unanswered());
46            }
47            Err(_elapsed) => {
48                log::warn!(target: "citadel", "[answer-wait] no answer for {what} within {within:?}; the subscription carried {carried} other event(s)");
49                return Err(unanswered());
50            }
51        }
52    }
53}

55#[cfg(test)]
56mod tests {
57    use super::*;
58    use citadel_io::tokio;
59    use citadel_io::ErrorCode;
60    use citadel_proto::prelude::{Disconnect, StackedRatchet, Ticket};
61
62    /// Short, so the deadline is reached quickly; the outer bound only stops a hang.
63    const WITHIN: Duration = Duration::from_millis(100);
64
65    fn event(success: bool) -> NodeResult<StackedRatchet> {
66        NodeResult::Disconnect(Disconnect {
67            ticket: Ticket(1),
68            cid_opt: Some(1),
69            success,
70            conn_type: None,
71            message: String::new(),
72            disconnect_token: None,
73            reason: None,
74        })
75    }
76
77    fn the_successful_one(event: NodeResult<StackedRatchet>) -> Option<Result<(), NetworkError>> {
78        matches!(
79            event,
80            NodeResult::Disconnect(Disconnect { success: true, .. })
81        )
82        .then_some(Ok(()))
83    }
84
85    #[tokio::test]
86    async fn a_request_nobody_answers_fails_at_its_deadline() {
87        let mut silent = futures::stream::pending::<NodeResult<StackedRatchet>>();
88        let outcome = tokio::time::timeout(
89            Duration::from_secs(10),
90            await_answer(
91                &mut silent,
92                WITHIN,
93                "test",
94                || citadel_io::error!(ErrorCode::RemoteDisconnectEventMissing),
95                the_successful_one,
96            ),
97        )
98        .await
99        .expect("the wait outlived its deadline");
100        assert_eq!(
101            outcome.unwrap_err().code(),
102            ErrorCode::RemoteDisconnectEventMissing
103        );
104    }
105
106    #[tokio::test]
107    async fn events_that_do_not_answer_are_skipped() {
108        let mut events = futures::stream::iter([event(false), event(false), event(true)]);
109        let outcome = await_answer(
110            &mut events,
111            WITHIN,
112            "test",
113            || citadel_io::error!(ErrorCode::RemoteDisconnectEventMissing),
114            the_successful_one,
115        )
116        .await;
117        assert!(outcome.is_ok(), "{outcome:?}");
118    }
119
120    #[tokio::test]
121    async fn a_subscription_that_ends_unanswered_is_reported() {
122        let mut events = futures::stream::iter([event(false)]);
123        let outcome = await_answer(
124            &mut events,
125            WITHIN,
126            "test",
127            || citadel_io::error!(ErrorCode::RemoteDeregisterEndedUnexpectedly),
128            the_successful_one,
129        )
130        .await;
131        assert_eq!(
132            outcome.unwrap_err().code(),
133            ErrorCode::RemoteDeregisterEndedUnexpectedly
134        );
135    }
